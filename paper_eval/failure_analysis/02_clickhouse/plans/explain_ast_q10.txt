SELECT\n    z.z_zonekey,\n    z.z_name AS pickup_zone,\n    AVG(dateDiff(\'second\', t.t_pickuptime, t.t_dropofftime)) AS avg_duration,\n    AVG(t.t_distance) AS avg_distance,\n    COUNT(t.t_tripkey) AS num_trips\nFROM zone AS z\nALL LEFT JOIN trip AS t ON geometryIntersectCartesian(readWKB(t.t_pickuploc), readWKB(z.z_boundary))\nGROUP BY\n    z.z_zonekey,\n    z.z_name\nORDER BY\n    avg_duration DESC NULLS LAST,\n    z.z_zonekey ASC\nLIMIT 100
