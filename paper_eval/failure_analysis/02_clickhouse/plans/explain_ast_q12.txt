WITH\n    trip_sampled AS\n    (\n        SELECT\n            t_tripkey,\n            readWKBPoint(t_pickuploc) AS pt\n        FROM trip\n    ),\n    buildings AS\n    (\n        SELECT\n            b_buildingkey,\n            readWKBPolygon(b_boundary) AS boundary\n        FROM building\n    )\nSELECT\n    t.t_tripkey,\n    AVG(b.dist) AS avg_distance_to_5_nearest\nFROM\n(\n    SELECT\n        t.t_tripkey,\n        polygonsDistanceCartesian(b.boundary, array(array(t.pt))) AS dist,\n        row_number() OVER (PARTITION BY t.t_tripkey ORDER BY dist ASC) AS rn\n    FROM trip_sampled AS t\n    CROSS JOIN buildings AS b\n) AS ranked\nWHERE lessOrEquals(rn, 5)\nGROUP BY t_tripkey\nORDER BY\n    avg_distance_to_5_nearest DESC,\n    t_tripkey ASC\nLIMIT 100
