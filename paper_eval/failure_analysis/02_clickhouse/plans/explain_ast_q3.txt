SELECT\n    toStartOfMonth(t.t_pickuptime) AS pickup_month,\n    COUNT(t.t_tripkey) AS total_trips,\n    AVG(t.t_distance) AS avg_distance,\n    AVG(dateDiff(\'second\', t.t_pickuptime, t.t_dropofftime)) AS avg_duration,\n    AVG(t.t_fare) AS avg_fare\nFROM trip AS t\nWHERE and(and(greaterOrEquals(tupleElement(readWKBPoint(t.t_pickuploc), 1), -111.951), lessOrEquals(tupleElement(readWKBPoint(t.t_pickuploc), 1), -111.571)), and(greaterOrEquals(tupleElement(readWKBPoint(t.t_pickuploc), 2), 34.6897), lessOrEquals(tupleElement(readWKBPoint(t.t_pickuploc), 2), 35.0497)))\nGROUP BY pickup_month\nORDER BY pickup_month ASC
