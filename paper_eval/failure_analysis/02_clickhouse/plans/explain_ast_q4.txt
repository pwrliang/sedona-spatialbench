SELECT\n    z.z_zonekey,\n    z.z_name,\n    COUNT(*) AS trip_count\nFROM\n(\n    SELECT\n        z_zonekey,\n        z_name,\n        readWKB(z_boundary) AS boundary\n    FROM zone\n) AS z\nALL INNER JOIN\n(\n    SELECT readWKB(t_pickuploc) AS pt\n    FROM trip\n    ORDER BY\n        t_tip DESC,\n        t_tripkey ASC\n    LIMIT 1000\n) AS top_trips ON geometryIntersectCartesian(top_trips.pt, z.boundary)\nGROUP BY\n    z.z_zonekey,\n    z.z_name\nORDER BY\n    trip_count DESC,\n    z.z_zonekey ASC
