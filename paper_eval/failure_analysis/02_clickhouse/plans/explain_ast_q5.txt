SELECT\n    c.c_custkey,\n    c.c_name AS customer_name,\n    toStartOfMonth(t.t_pickuptime) AS pickup_month,\n    abs(polygonAreaCartesian(polygonConvexHullCartesian(groupArray(readWKBPoint(t.t_dropoffloc))))) AS monthly_travel_hull_area,\n    COUNT(*) AS dropoff_count\nFROM trip AS t\nALL INNER JOIN customer AS c ON equals(t.t_custkey, c.c_custkey)\nGROUP BY\n    c.c_custkey,\n    c.c_name,\n    pickup_month\nHAVING greater(dropoff_count, 5)\nORDER BY\n    monthly_travel_hull_area DESC,\n    c.c_custkey ASC,\n    pickup_month ASC\nLIMIT 100
