WITH\n    filtered_zones AS\n    (\n        SELECT\n            z_zonekey,\n            z_name,\n            readWKB(z_boundary) AS boundary\n        FROM zone\n        WHERE geometryIntersectCartesian(readWKT(\'POLYGON((-112.2110 34.4197, -111.3110 34.4197, -111.3110 35.3197, -112.2110 35.3197, -112.2110 34.4197))\'), readWKB(z_boundary))\n    )\nSELECT\n    z.z_zonekey,\n    z.z_name,\n    COUNT(t.t_tripkey) AS total_pickups,\n    AVG(t.t_distance) AS avg_distance,\n    AVG(dateDiff(\'second\', t.t_pickuptime, t.t_dropofftime)) AS avg_duration\nFROM trip AS t\nALL INNER JOIN filtered_zones AS z ON geometryIntersectCartesian(readWKB(t.t_pickuploc), z.boundary)\nGROUP BY\n    z.z_zonekey,\n    z.z_name\nORDER BY\n    total_pickups DESC,\n    z.z_zonekey ASC
