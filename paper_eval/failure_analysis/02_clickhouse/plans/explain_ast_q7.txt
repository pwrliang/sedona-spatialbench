WITH\n    trip_lengths AS\n    (\n        SELECT\n            t.t_tripkey,\n            t.t_distance AS reported_distance_m,\n            divide(L2Distance(readWKBPoint(t.t_pickuploc), readWKBPoint(t.t_dropoffloc)), 0.000009) AS line_distance_m\n        FROM trip AS t\n    )\nSELECT\n    t.t_tripkey,\n    t.reported_distance_m,\n    t.line_distance_m,\n    if(equals(t.line_distance_m, 0), _CAST(NULL, \'Nullable(Nothing)\'), divide(t.reported_distance_m, t.line_distance_m)) AS detour_ratio\nFROM trip_lengths AS t\nORDER BY\n    detour_ratio DESC NULLS LAST,\n    reported_distance_m DESC,\n    t_tripkey ASC\nLIMIT 100
