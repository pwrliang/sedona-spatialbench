WITH\n    b1 AS\n    (\n        SELECT\n            b_buildingkey AS id,\n            readWKBPolygon(b_boundary) AS geom\n        FROM building\n    ),\n    b2 AS\n    (\n        SELECT\n            b_buildingkey AS id,\n            readWKBPolygon(b_boundary) AS geom\n        FROM building\n    ),\n    pairs AS\n    (\n        SELECT\n            b1.id AS building_1,\n            b2.id AS building_2,\n            abs(polygonAreaCartesian(b1.geom)) AS area1,\n            abs(polygonAreaCartesian(b2.geom)) AS area2,\n            abs(polygonAreaCartesian(polygonsIntersectionCartesian(b1.geom, b2.geom))) AS overlap_area\n        FROM b1\n        ALL INNER JOIN b2 ON and(less(b1.id, b2.id), polygonsIntersectCartesian(b1.geom, b2.geom))\n    )\nSELECT\n    building_1,\n    building_2,\n    area1,\n    area2,\n    overlap_area,\n    if(equals(minus(plus(area1, area2), overlap_area), 0), 0., divide(overlap_area, minus(plus(area1, area2), overlap_area))) AS iou\nFROM pairs\nORDER BY\n    iou DESC,\n    building_1 ASC,\n    building_2 ASC\nLIMIT 100
